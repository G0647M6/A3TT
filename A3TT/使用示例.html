🎯 选衣录核心算法库 - 使用示例

引用 xuanyilu-core.js 后，通过 XuanYiLuCore 对象调用全部算法功能。

📦 引入方式

<script src="xuanyilu-core.js"></script>

🎨 颜色提取示例

// 从 canvas 提取颜色（自动排除背景和阴影）
var colors = XuanYiLuCore.extractColors(canvasImageData);
// 返回: { main: '#HEX', secondary: '#HEX', accent: '#HEX', palette: [...] }

👗 生成搭配示例

// 设置用户信息
XuanYiLuCore.setGender('female');
XuanYiLuCore.setAge('worker');
XuanYiLuCore.setOccasion('date');
XuanYiLuCore.setSeason('spring');

// 传入衣服列表，生成 6 套搭配
var results = XuanYiLuCore.generateCombinations(items, 6);

🔬 色彩工具

// RGB 转 HSL
var hsl = XuanYiLuCore.rgbToHsl(255, 100, 100);

// 计算色相距离（环式，0-180°）
var dist = XuanYiLuCore.hueDistance(0, 30);

// 判断是否中性色（黑白灰米）
var isGray = XuanYiLuCore.isNeutral('#888888');

🤖 大模型接口

// 检查大模型是否配置好
if (XuanYiLuCore.isAIOutfitReady()) {
  // 构建上下文
  var ctx = XuanYiLuCore.buildAIContext(items);
  // 调用大模型生成搭配
  XuanYiLuCore.callAIOutfit(items, 6).then(function(result) {
    console.log(result);
  });
}
